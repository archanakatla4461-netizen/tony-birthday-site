<script>
function showScreen(screenId) {
document.querySelectorAll(".screen").forEach(function(screen) {
screen.classList.remove("active");
});

document.getElementById(screenId).classList.add("active");
window.scrollTo(0, 0);
}

function uploadPhoto(event, imageId) {
const file = event.target.files[0];

if (file && file.type.startsWith("image/")) {
const reader = new FileReader();

reader.onload = function() {
document.getElementById(imageId).src = reader.result;
};

reader.readAsDataURL(file);
}
}

function openGift() {
document.getElementById("giftBox").textContent = "🎉";
document.getElementById("giftHint").textContent = "Your surprise is open! 💗";
document.getElementById("giftMessage").classList.remove("hidden");
}

// Optional music: press the button to start or stop it.
let audioContext = null;
let musicTimer = null;
let noteIndex = 0;

const notes = [523.25, 659.25, 783.99, 659.25, 587.33, 698.46, 880, 698.46];

function playNote() {
if (!audioContext) return;

const oscillator = audioContext.createOscillator();
const gain = audioContext.createGain();

oscillator.type = "sine";
oscillator.frequency.value = notes[noteIndex % notes.length];

gain.gain.setValueAtTime(0.0001, audioContext.currentTime);
gain.gain.exponentialRampToValueAtTime(0.06, audioContext.currentTime + 0.05);
gain.gain.exponentialRampToValueAtTime(0.0001, audioContext.currentTime + 0.35);

oscillator.connect(gain);
gain.connect(audioContext.destination);

oscillator.start();
oscillator.stop(audioContext.currentTime + 0.4);

noteIndex++;
}

function toggleMusic() {
const button = document.getElementById("musicButton");

if (musicTimer) {
clearInterval(musicTimer);
musicTimer = null;
button.textContent = "Play Music 🎵";
if (audioContext) audioContext.suspend();
return;
}

const AudioContextClass = window.AudioContext || window.webkitAudioContext;

if (!AudioContextClass) {
alert("Music is not supported in this browser.");
return;
}

if (!audioContext) audioContext = new AudioContextClass();

audioContext.resume();
playNote();
musicTimer = setInterval(playNote, 450);
button.textContent = "Stop Music 🔇";
}
